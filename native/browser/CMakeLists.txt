option(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER "Build Debug-only macOS no-exec fronts and bodies for patched CEF" OFF)
if(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER)
  if(NOT OS_MAC OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR CMAKE_CONFIGURATION_TYPES OR NOT COLOSSUS_CEF_DEVELOPER_NETWORK_ENVELOPE)
    message(FATAL_ERROR "Owned process lane requires explicit macOS Debug and network envelope; no production acceptance.")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    set(_owned_platform "macosarm64")
  else()
    set(_owned_platform "macosx64")
  endif()
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/scripts/owned_process_component.py"
    --cef-root "${CEF_ROOT}" --platform "${_owned_platform}"
    --write-build "${CMAKE_CURRENT_BINARY_DIR}"
    RESULT_VARIABLE _owned_validation)
  if(NOT _owned_validation EQUAL 0)
    message(FATAL_ERROR "Owned process lane requires a verified source-built CEF with actual private loader exports; stock CEF is unsupported.")
  endif()
  target_sources(colossus_cef PRIVATE process-broker/main_entry.cc)
  target_compile_definitions(colossus_cef PRIVATE COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER=1)
endif()
if(COLOSSUS_CEF_MAC_SYNTHETIC_PROFILE_DIAGNOSTIC)
  if(NOT OS_MAC OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR COLOSSUS_CEF_DEVELOPER_PROFILE_CRYPTO)
    message(FATAL_ERROR "The synthetic profile diagnostic requires macOS Debug and excludes the real profile adapter; no production custody is enabled.")
  endif()
  target_compile_definitions(colossus_cef PRIVATE COLOSSUS_CEF_MAC_SYNTHETIC_PROFILE_DIAGNOSTIC=1)
  if(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/synthetic-profile-diagnostic-build.json"
      "{\"schema_version\":1,\"development_only\":true,\"build_type\":\"Debug\",\"production_accepted\":false,\"api\":\"chromium-use-mock-keychain\",\"loopback_only\":true,\"https_broker_leaf_pin_required\":true,\"temporary_profiles_only\":true}\n")
  else()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/synthetic-profile-diagnostic-build.json"
      "{\"schema_version\":1,\"development_only\":true,\"build_type\":\"Debug\",\"production_accepted\":false,\"api\":\"chromium-use-mock-keychain\",\"http_loopback_only\":true,\"temporary_profiles_only\":true}\n")
  endif()
else()
  file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/synthetic-profile-diagnostic-build.json")
endif()
if(COLOSSUS_CEF_DEVELOPER_NETWORK_ENVELOPE)
  if(NOT OS_MAC OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR
      (NOT COLOSSUS_CEF_DEVELOPER_PROFILE_CRYPTO AND NOT COLOSSUS_CEF_MAC_SYNTHETIC_PROFILE_DIAGNOSTIC))
    message(FATAL_ERROR "The macOS network envelope requires an explicit Debug owned-profile or synthetic fixture build; no production capability is enabled.")
  endif()
  include(mac/network_envelope.cmake)
  if(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER)
    include(mac/owned_process.cmake)
    add_executable(colossus-mac-browser-envelope-front process-broker/front_loader.c)
    target_compile_definitions(colossus-mac-browser-envelope-front PRIVATE COLOSSUS_OWNED_MAIN_FRONT=1)
    target_link_libraries(colossus-mac-browser-envelope-front PRIVATE colossus_mac_owned_policy sandbox)
  else()
    add_executable(colossus-mac-browser-envelope-front mac/host_envelope_front.c)
    target_link_libraries(colossus-mac-browser-envelope-front PRIVATE colossus_mac_network_policy sandbox)
  endif()
  target_compile_options(colossus-mac-browser-envelope-front PRIVATE -Wall -Wextra -Werror)
  set_target_properties(colossus-mac-browser-envelope-front PROPERTIES
    OUTPUT_NAME "colossus-native-browser-host"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/envelope")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/network-envelope-development-build.json"
    "{\"schema_version\":1,\"development_only\":true,\"build_type\":\"Debug\",\"production_accepted\":false,\"native_acceptance\":false}\n")
endif()
if(COLOSSUS_CEF_DEVELOPER_PROFILE_CRYPTO)
  if(NOT OS_MAC OR NOT CMAKE_BUILD_TYPE STREQUAL "Debug")
    message(FATAL_ERROR "Owned profile crypto requires an explicit macOS Debug development build; it is not production acceptance.")
  endif()
  # dyld ignores interpose sections in MH_EXECUTE. This must remain a normal
  # signed MH_DYLIB launch dependency, never a merged/static library or injection.
  add_library(colossus_mac_profile_crypto SHARED mac/profile_crypto.mm)
  target_compile_definitions(colossus_mac_profile_crypto PRIVATE COLOSSUS_MAC_PROFILE_CRYPTO_DEVELOPMENT=1)
  target_include_directories(colossus_mac_profile_crypto PUBLIC mac)
  target_link_libraries(colossus_mac_profile_crypto PRIVATE "-framework Security" "-framework CoreFoundation")
  target_compile_options(colossus_mac_profile_crypto PRIVATE -Wall -Wextra -Werror)
  set_target_properties(colossus_mac_profile_crypto PROPERTIES
    MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/profile-crypto-development-build.json"
    "{\"schema_version\":1,\"development_only\":true,\"build_type\":\"Debug\",\"production_accepted\":false,\"api\":\"unsupported-dyld-security-spi\"}\n")
  install(TARGETS colossus_mac_profile_crypto LIBRARY DESTINATION .)
  add_executable(colossus-mac-profile-crypto-ownership-test EXCLUDE_FROM_ALL mac/profile_crypto_ownership_test.mm)
  target_compile_definitions(colossus-mac-profile-crypto-ownership-test PRIVATE COLOSSUS_MAC_PROFILE_CRYPTO_DEVELOPMENT=1)
  target_link_libraries(colossus-mac-profile-crypto-ownership-test PRIVATE "-framework Security" "-framework CoreFoundation")
  target_compile_options(colossus-mac-profile-crypto-ownership-test PRIVATE -Wall -Wextra -Werror)
  # Explicit operator fixture only. Building/signing it does not run Security
  # operations; use the reviewed two-step runner and fresh private evidence.
  add_executable(colossus-mac-profile-crypto-fixture EXCLUDE_FROM_ALL mac/profile_crypto_fixture.mm)
  target_include_directories(colossus-mac-profile-crypto-fixture PRIVATE mac)
  target_link_libraries(colossus-mac-profile-crypto-fixture PRIVATE colossus_mac_profile_crypto "-framework Security" "-framework CoreFoundation")
  target_compile_options(colossus-mac-profile-crypto-fixture PRIVATE -Wall -Wextra -Werror)
endif()

if(OS_MAC)
  target_sources(colossus_cef PRIVATE src/platform_mac.mm src/mac_helper_programs.cc)
  target_link_libraries(colossus_cef PUBLIC "-framework Cocoa")
  # Pure pre-bootstrap configuration checks: this executable never loads CEF,
  # initializes AppKit, runs a browser, or calls any Keychain operation.
  add_executable(colossus-cef-synthetic-profile-test EXCLUDE_FROM_ALL tests/synthetic_profile_test.cc)
  target_include_directories(colossus-cef-synthetic-profile-test PRIVATE src)
  target_link_libraries(colossus-cef-synthetic-profile-test PRIVATE colossus_cef)
  target_compile_options(colossus-cef-synthetic-profile-test PRIVATE -Wall -Wextra -Werror)
  if(COLOSSUS_CEF_MAC_SYNTHETIC_PROFILE_DIAGNOSTIC)
    target_compile_definitions(colossus-cef-synthetic-profile-test PRIVATE COLOSSUS_EXPECT_SYNTHETIC_PROFILE=1)
  endif()
  add_executable(colossus-mac-dedicated-environment-test EXCLUDE_FROM_ALL tests/mac_dedicated_environment_test.cc)
  target_include_directories(colossus-mac-dedicated-environment-test PRIVATE src)
  target_link_libraries(colossus-mac-dedicated-environment-test PRIVATE colossus_cef)
  target_compile_options(colossus-mac-dedicated-environment-test PRIVATE -Wall -Wextra -Werror)
  add_executable(colossus-mac-helper-programs-test EXCLUDE_FROM_ALL
    tests/mac_helper_programs_test.cc src/mac_helper_programs.cc)
  target_include_directories(colossus-mac-helper-programs-test PRIVATE src)
  target_compile_options(colossus-mac-helper-programs-test PRIVATE -Wall -Wextra -Werror)
  # CEF selects typed helpers by adding these pinned suffixes to the base
  # subprocess bundle. Keep each bundle/executable and identifier distinct.
  foreach(_suffix_list ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" _suffix_list "${_suffix_list}")
    list(GET _suffix_list 0 _name_suffix)
    list(GET _suffix_list 1 _target_suffix)
    list(GET _suffix_list 2 _plist_suffix)
    set(_helper_target "colossus-browser-helper${_target_suffix}")
    set(COLOSSUS_HELPER_NAME "Colossus Browser Helper${_name_suffix}")
    set(COLOSSUS_HELPER_IDENTIFIER "com.obscuritylabs.colossus.chromium-preview.helper${_plist_suffix}")
    set(_helper_plist "${CMAKE_CURRENT_BINARY_DIR}/helper${_target_suffix}-Info.plist")
    configure_file(mac/helper-Info.plist.in "${_helper_plist}" @ONLY)
    if(COLOSSUS_CEF_DEVELOPER_NETWORK_ENVELOPE)
      if(_target_suffix STREQUAL "_renderer")
        set(_helper_kind 1)
      elseif(_target_suffix STREQUAL "_gpu")
        set(_helper_kind 2)
      elseif(_target_suffix STREQUAL "_plugin")
        set(_helper_kind 3)
      elseif(_target_suffix STREQUAL "_alerts")
        set(_helper_kind 4)
      else()
        set(_helper_kind 0)
      endif()
      if(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER)
        colossus_add_owned_helper(${_helper_target} "${COLOSSUS_HELPER_NAME}" "${_helper_plist}")
      else()
        colossus_add_network_helper(${_helper_target} "${COLOSSUS_HELPER_NAME}"
          "${_helper_plist}" "${_helper_kind}")
      endif()
    else()
      add_executable(${_helper_target} MACOSX_BUNDLE src/helper_mac.mm)
      SET_EXECUTABLE_TARGET_PROPERTIES(${_helper_target})
      target_include_directories(${_helper_target} PRIVATE "${CEF_ROOT}")
      target_link_libraries(${_helper_target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
      set_target_properties(${_helper_target} PROPERTIES
        OUTPUT_NAME "${COLOSSUS_HELPER_NAME}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/helpers"
        MACOSX_BUNDLE_INFO_PLIST "${_helper_plist}")
      install(TARGETS ${_helper_target} BUNDLE DESTINATION helpers)
    endif()
  endforeach()
  if(COLOSSUS_CEF_DEVELOPER_PROCESS_BROKER)
    colossus_add_owned_artifact_receipt()
  endif()
  # The distribution framework is unversioned. The development app stager
  # creates the supported Versions/A layout and signs only copied bytes.
  install(DIRECTORY "${CEF_ROOT}/Release/Chromium Embedded Framework.framework"
    DESTINATION Frameworks USE_SOURCE_PERMISSIONS)
endif()

endif()
