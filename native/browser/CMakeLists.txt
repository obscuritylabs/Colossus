cmake_minimum_required(VERSION 3.21)
project(colossus_browser LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()

if(NOT IS_DIRECTORY "${CEF_ROOT}")
  message(FATAL_ERROR "Set CEF_ROOT to the verified pinned minimal distribution.")
endif()
file(READ "${CEF_ROOT}/include/cef_version.h" COLOSSUS_CEF_VERSION_HEADER)
if(NOT COLOSSUS_CEF_VERSION_HEADER MATCHES
  "#define CEF_VERSION \"154[.]0[.]34[+]g14c5a08[+]chromium-154[.]0[.]8037[.]98\"")
  message(FATAL_ERROR "CEF_ROOT must match native/browser/component/cef.lock.json.")
endif()
if(DEFINED USE_SANDBOX AND NOT USE_SANDBOX)
  message(FATAL_ERROR "Colossus browser requires the Chromium process sandbox.")
endif()
set(USE_SANDBOX ON CACHE BOOL "Mandatory Chromium sandbox" FORCE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
if(NOT OS_MAC)
  ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
endif()

add_library(colossus_cef STATIC src/host.cc src/client.cc src/control.cc src/platform.cc src/presentation.cc src/transfer.cc src/profile.cc)
option(COLOSSUS_CEF_TEST_CUSTODY "Build the closed Linux native renderer NSS open-denial experiment" OFF)
if(COLOSSUS_CEF_TEST_CUSTODY)
  if(NOT OS_LINUX)
    message(FATAL_ERROR "The native custody experiment requires the pinned Linux renderer sandbox.")
  endif()
  target_sources(colossus_cef PRIVATE src/custody_linux.cc src/custody_proc_linux.cc)
  target_compile_definitions(colossus_cef PRIVATE COLOSSUS_CEF_TEST_CUSTODY=1)
endif()
if(OS_WINDOWS)
  target_sources(colossus_cef PRIVATE src/windows_entry.cc src/platform_win.cc)
endif()
target_include_directories(colossus_cef PUBLIC include PRIVATE "${CEF_ROOT}")
SET_LIBRARY_TARGET_PROPERTIES(colossus_cef)
target_link_libraries(colossus_cef PUBLIC libcef_dll_wrapper)
if(NOT OS_LINUX)
  target_link_libraries(colossus_cef PUBLIC ${CEF_STANDARD_LIBS})
endif()
if(NOT OS_MAC)
  target_link_libraries(colossus_cef PUBLIC libcef_lib)
endif()
if(OS_WINDOWS)
  add_library(colossus-browser-helper SHARED src/helper_win.cc)
  SET_LIBRARY_TARGET_PROPERTIES(colossus-browser-helper)
  target_include_directories(colossus-browser-helper PRIVATE include "${CEF_ROOT}")
  target_link_libraries(colossus-browser-helper PRIVATE colossus_cef)
  install(TARGETS colossus-browser-helper RUNTIME DESTINATION .)
endif()
if(OS_WINDOWS OR OS_MAC)
  if(OS_WINDOWS)
    add_library(colossus_browser_presenter STATIC src/presenter_win.cc)
    target_link_libraries(colossus_browser_presenter PUBLIC imm32)
    add_executable(colossus-browser-presenter-probe src/presenter_probe_win.cc)
    target_link_libraries(colossus-browser-presenter-probe PRIVATE user32 gdi32)
  else()
    add_library(colossus_browser_presenter STATIC src/presenter_mac.mm)
    target_link_libraries(colossus_browser_presenter PUBLIC "-framework Cocoa")
    add_executable(colossus-browser-presenter-probe src/presenter_probe_mac.mm)
  endif()
  SET_LIBRARY_TARGET_PROPERTIES(colossus_browser_presenter)
  target_include_directories(colossus_browser_presenter PUBLIC include)
  target_link_libraries(colossus-browser-presenter-probe PRIVATE colossus_browser_presenter)
  install(TARGETS colossus_browser_presenter ARCHIVE DESTINATION lib)
endif()
if(OS_MAC)
  target_sources(colossus_cef PRIVATE src/platform_mac.mm)
  target_link_libraries(colossus_cef PUBLIC "-framework Cocoa")
  # CEF selects typed helpers by adding these pinned suffixes to the base
  # subprocess bundle. Keep each bundle/executable and identifier distinct.
  foreach(_suffix_list ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" _suffix_list "${_suffix_list}")
    list(GET _suffix_list 0 _name_suffix)
    list(GET _suffix_list 1 _target_suffix)
    list(GET _suffix_list 2 _plist_suffix)
    set(_helper_target "colossus-browser-helper${_target_suffix}")
    set(COLOSSUS_HELPER_NAME "Colossus Browser Helper${_name_suffix}")
    set(COLOSSUS_HELPER_IDENTIFIER "com.obscuritylabs.colossus.chromium-preview.helper${_plist_suffix}")
    set(_helper_plist "${CMAKE_CURRENT_BINARY_DIR}/helper${_target_suffix}-Info.plist")
    configure_file(mac/helper-Info.plist.in "${_helper_plist}" @ONLY)
    add_executable(${_helper_target} MACOSX_BUNDLE src/helper_mac.mm)
    SET_EXECUTABLE_TARGET_PROPERTIES(${_helper_target})
    target_include_directories(${_helper_target} PRIVATE "${CEF_ROOT}")
    target_link_libraries(${_helper_target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    set_target_properties(${_helper_target} PROPERTIES
      OUTPUT_NAME "${COLOSSUS_HELPER_NAME}"
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/helpers"
      MACOSX_BUNDLE_INFO_PLIST "${_helper_plist}")
    install(TARGETS ${_helper_target} BUNDLE DESTINATION helpers)
  endforeach()
  # The distribution framework is unversioned. The development app stager
  # creates the supported Versions/A layout and signs only copied bytes.
  install(DIRECTORY "${CEF_ROOT}/Release/Chromium Embedded Framework.framework"
    DESTINATION Frameworks USE_SOURCE_PERMISSIONS)
endif()

# Probe proves only Linux no-display startup and direct method continuity. It does
# not promote a release capability or bypass required sandbox/network acceptance.
if(OS_LINUX)
  add_executable(colossus-browser-probe src/probe.cc)
  target_include_directories(colossus-browser-probe PRIVATE "${CEF_ROOT}")
  SET_EXECUTABLE_TARGET_PROPERTIES(colossus-browser-probe)
  target_link_libraries(colossus-browser-probe PRIVATE colossus_cef)
  set_target_properties(colossus-browser-probe PROPERTIES
    BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  install(TARGETS colossus-browser-probe RUNTIME DESTINATION .)
  set(COLOSSUS_BROWSER_HOST_EXECUTABLE "" CACHE FILEPATH
    "Separately built private Rust native browser host to stage beside CEF")
  if(COLOSSUS_BROWSER_HOST_EXECUTABLE)
    if(NOT EXISTS "${COLOSSUS_BROWSER_HOST_EXECUTABLE}")
      message(FATAL_ERROR "COLOSSUS_BROWSER_HOST_EXECUTABLE must exist before staging.")
    endif()
    get_filename_component(_host_name "${COLOSSUS_BROWSER_HOST_EXECUTABLE}" NAME)
    if(NOT _host_name STREQUAL "colossus-native-browser-host")
      message(FATAL_ERROR "Only the private native browser host may be staged here.")
    endif()
    install(PROGRAMS "${COLOSSUS_BROWSER_HOST_EXECUTABLE}" DESTINATION .)
  endif()
  install(DIRECTORY "${CEF_ROOT}/Release/" DESTINATION .
    PATTERN "*.a" EXCLUDE PATTERN "*.lib" EXCLUDE)
  install(DIRECTORY "${CEF_ROOT}/Resources/" DESTINATION .)
  # CEF initializes this directory beside its executable even when extensions
  # are disabled. Package its empty immutable location before inventorying.
  install(DIRECTORY DESTINATION extensions DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE)
endif()
install(TARGETS colossus_cef ARCHIVE DESTINATION lib)
install(TARGETS libcef_dll_wrapper ARCHIVE DESTINATION lib)
install(FILES include/colossus_cef.h include/colossus_cef_presentation.h
  include/colossus_browser_presenter.h include/colossus_cef_transfer.h DESTINATION include)
install(FILES "${CEF_ROOT}/LICENSE.txt" "${CEF_ROOT}/CREDITS.html" DESTINATION notices)
